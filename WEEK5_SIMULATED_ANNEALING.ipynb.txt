        "        neighbor[col] = new_row\n",
        "        return col, new_row, neighbor\n",
        "\n",
        "    def run_simulation(self, initial_temp=10.0, alpha=0.995, max_iterations=10000):\n",
        "        current_board = list(self.initial_state)\n",
        "        current_cost = self.calculate_cost(current_board)\n",
        "        T = initial_temp\n",
        "        iteration = 0\n",
        "\n",
        "        print(\"\\n=== STARTING SIMULATED ANNEALING ===\")\n",
        "        print(f\"Initial Configuration: {current_board} | Cost: {current_cost} conflicts\\n\")\n",
        "\n",
        "        while T > 0.0001 and current_cost > 0 and iteration < max_iterations:\n",
        "            iteration += 1\n",
        "            col, new_row, proposed_board = self.get_random_neighbor(current_board)\n",
        "            proposed_cost = self.calculate_cost(proposed_board)\n",
        "\n",
        "            # 4. Acceptance Criterion: ΔE = Cost_new - Cost_current\n",
        "            delta_E = proposed_cost - current_cost\n",
        "\n",
        "            # Case 1: ΔE < 0 (Better Move) -> Always Accept\n",
        "            if delta_E < 0:\n",
        "                current_board = proposed_board\n",
        "                current_cost = proposed_cost\n",
        "                print(f\"[Iter {iteration:03d}] Better Move -> Board: {current_board} | Cost: {current_cost} | T: {T:.4f}\")\n",
        "\n",
        "            # Case 2: ΔE >= 0 (Worse Move) -> Accept Probabilistically\n",
        "            else:\n",
        "                p = math.exp(-delta_E / T)\n",
        "                if random.random() < p:\n",
        "                    current_board = proposed_board\n",
        "                    current_cost = proposed_cost\n",
        "                    print(f\"[Iter {iteration:03d}] Worse Move Accepted (P={p:.3f}) -> Board: {current_board} | Cost: {current_cost} | T: {T:.4f}\")\n",
        "\n",
        "            # 5. Cooling Schedule: T_(k+1) = T_k * alpha\n",
        "            T *= alpha\n",
        "\n",
        "        print(\"\\n=== CONVERGENCE SUMMARY ===\")\n",
        "        if current_cost == 0:\n",
        "            print(f\"🎉 SUCCESS! Found goal state: {current_board}\")\n",
        "        else:\n",
        "            print(f\"🛑 Stopped at Local Minimum: {current_board} | Remaining Cost: {current_cost}\")\n",
        "\n",
        "        self.print_board(current_board)\n",
        "\n",
